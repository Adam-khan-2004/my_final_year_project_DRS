# ==========================================================
# BALL DETECTION
# ==========================================================

@app.route("/ball", methods=["GET", "POST"])
def ball():

    if request.method == "POST":

        if "video" not in request.files:
            return render_template(
                "ball.html",
                result="No video selected."
            )

        video = request.files["video"]

        if video.filename == "":
            return render_template(
                "ball.html",
                result="No video selected."
            )

        # ------------------------------------------
        # Save Uploaded Video
        # ------------------------------------------

        upload_path = os.path.join(
            app.config["UPLOAD_FOLDER"],
            video.filename
        )

        video.save(upload_path)

        print("\n==============================")
        print("Uploaded Video")
        print(upload_path)
        print("==============================")

        # ------------------------------------------
        # Run Ball Detection
        # ------------------------------------------

        detected_video, result = detect_ball(upload_path)

        print("\n==============================")
        print("Returned From ball.py")
        print("Detected Video :", detected_video)
        print("Result :", result)
        print("==============================")

        # ------------------------------------------
        # Render Page
        # ------------------------------------------

        return render_template(

            "ball.html",

            result=result,

            uploaded_video=url_for(
                "static",
                filename="uploads/" + video.filename
            ),

            detected_video=url_for(
                "static",
                filename=detected_video
            )
        )

    # GET Request

    return render_template(
        "ball.html"
    )
